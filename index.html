// [신규 추가] 버튼 클릭 시 AI 서버로 데이터를 요청하고 결과를 받아오는 함수입니다.
async function generateVBA() {
    // 1. 사용자가 입력한 요청 사항을 가져옵니다.
    const userText = document.getElementById('userInput').value;
    
    // 2. 입력값이 없으면 경고창을 띄우고 함수를 종료합니다.
    if (!userText) {
        alert("원하는 기능을 입력해주세요.");
        return;
    }

    // 결과 영역 초기화 및 로딩 표시
    document.getElementById('vbaCode').innerText = "AI가 코드와 디자인을 생성 중입니다... 잠시만 기다려주세요.";
    document.getElementById('uiImage').style.display = "none";

    try {
        // 3. 백엔드 서버(혹은 직접 AI API)로 요청을 보냅니다.
        // *주의: 실제 구현 시에는 보안을 위해 백엔드 서버(Node.js, Python 등)를 거쳐 API 키를 보호해야 합니다.
        const response = await fetch('https://내-서버-주소.com/api/generate', {
            method: 'POST', // 데이터를 서버로 보낼 때 사용하는 방식
            headers: {
                'Content-Type': 'application/json' // 주고받는 데이터 형식을 JSON으로 지정
            },
            // 사용자의 입력을 JSON 형태로 변환하여 전송
            body: JSON.stringify({ prompt: userText }) 
        });

        // 4. 서버로부터 받은 응답 데이터를 JSON 형식으로 변환합니다.
        const data = await response.json();

        // 5. 서버에서 받아온 VBA 코드를 화면에 표시합니다.
        document.getElementById('vbaCode').innerText = data.vba_code;
        
        // 6. 서버에서 받아온 UI 이미지 URL을 이미지 태그에 넣고 화면에 보이게 합니다.
        document.getElementById('uiImage').src = data.image_url;
        document.getElementById('uiImage').style.display = "block";

    } catch (error) {
        // 에러 발생 시 처리
        console.error("생성 중 오류 발생:", error);
        document.getElementById('vbaCode').innerText = "데이터를 불러오는 중 오류가 발생했습니다.";
    }
}
